#!/usr/bin/env python3
"""
Simple Grok memory manager.
Run with: python memory.py
"""

import os
from datetime import datetime

MEMORY_FILE = "grok_memory.md"

def load_memory():
    if not os.path.exists(MEMORY_FILE):
        return ""
    with open(MEMORY_FILE, "r", encoding="utf-8") as f:
        return f.read()

def save_memory(content):
    with open(MEMORY_FILE, "w", encoding="utf-8") as f:
        f.write(content)

def add_note(note: str):
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M")
    entry = f"\n- {timestamp}: {note}"
    
    content = load_memory()
    if "## Session Log" not in content:
        content += "\n\n## Session Log\n"
    
    # Insert after the Session Log header
    parts = content.split("## Session Log")
    if len(parts) == 2:
        content = parts[0] + "## Session Log" + parts[1] + entry
    else:
        content += entry
    
    save_memory(content)
    print("Note added.")

def generate_context(max_chars=8000):
    content = load_memory()
    if len(content) > max_chars:
        print(f"Warning: Memory is long ({len(content)} chars). Consider trimming older session logs.")
    print("\n" + "="*60)
    print("COPY THE BLOCK BELOW INTO A NEW GROK CHAT")
    print("="*60 + "\n")
    print(content)
    print("\n" + "="*60)
    print("END OF CONTEXT BLOCK")
    print("="*60)

def main():
    while True:
        print("\nGrok Memory Manager")
        print("1. Add a note / update")
        print("2. Show full memory")
        print("3. Generate context block to paste")
        print("4. Exit")
        choice = input("\nChoice: ").strip()

        if choice == "1":
            note = input("Note: ").strip()
            if note:
                add_note(note)
        elif choice == "2":
            print("\n" + load_memory())
        elif choice == "3":
            generate_context()
        elif choice == "4":
            break
        else:
            print("Invalid choice.")

if __name__ == "__main__":
    main()

def read_file(path: str) -> str:
    """Lee el contenido de un archivo. Recibe la ruta del archivo."""
    try:
        with open(path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        return f"Error: '{path}' no existe."
    except PermissionError:
        return f"Error: Sin permiso para leer '{path}'."

def write_file(path: str, content: str) -> str:
    """Escribe contenido en un archivo (sobrescribe si existe). Recibe ruta y contenido."""
    try:
        os.makedirs(os.path.dirname(path) or ".", exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            f.write(content)
        return f"✅ Escrito: {path} ({len(content)} chars)"
    except Exception as e:
        return f"❌ Error escribiendo '{path}': {e}"

def list_files(directory: str = ".") -> str:
    """Lista los archivos y carpetas en un directorio. Recibe la ruta (default: '.')."""
    try:
        entries = os.listdir(directory)
        resultado = []
        for e in sorted(entries):
            full = os.path.join(directory, e)
            tipo = "📁" if os.path.isdir(full) else "📄"
            resultado.append(f"{tipo} {e}")
        return "\n".join(resultado) if resultado else "(vacío)"
    except Exception as e:
        return f"❌ Error: {e}"

def run_code(path: str) -> str:
    """
    Ejecuta un archivo Python y devuelve el output (stdout + stderr).
    Recibe la ruta del archivo .py a ejecutar.
    Úsalo para PROBAR el código que acabas de escribir.
    """
    try:
        result = subprocess.run(
            [sys.executable, path],
            capture_output=True,
            text=True,
            timeout=30
        )
        output = ""
        if result.stdout:
            output += f"STDOUT:\n{result.stdout}\n"
        if result.stderr:
            output += f"STDERR:\n{result.stderr}\n"
        if result.returncode == 0:
            return f"✅ Exit code 0\n{output}"
        else:
            return f"❌ Exit code {result.returncode}\n{output}"
    except subprocess.TimeoutExpired:
        return "❌ Timeout: el código tardó más de 30 segundos."
    except Exception as e:
        return f"❌ Error ejecutando: {e}"

def rag_query(question: str) -> str:
    """
    Busca información en tus documentos indexados (RAG/ChromaDB).
    Úsalo cuando necesites contexto de tus apuntes o documentos.
    Recibe la pregunta en texto natural.
    """
    try:
        from rag import query_engine
        return str(query_engine.query(question))
    except ImportError:
        return "❌ No se pudo importar rag.py. Asegúrate de que esté en la misma carpeta."
    except Exception as e:
        return f"❌ Error en RAG: {e}"

# =================================================

# =================================================

def get_existing_files() -> str:
    """Obtiene la lista de archivos actuales para inyectar en el prompt."""
    try:
        entries = sorted(os.listdir("."))
        return "\n".join(f" - {e}" for e in entries)
    except:
        return " (no se pudo listar)"

SYSTEM_PROMPT = """Eres un agente especializado en crear otros agentes de Python.
Recibes una ESPECIFICACIÓN APROBADA por el usuario y la conviertes en código funcional.

REGLAS:
1. Solo escribes código cuando recibes una especificación clara y aprobada.
2. Antes de modificar un archivo existente, LÉELO primero con read_file.
3. Después de escribir un archivo .py, EJECÚTALO con run_code para verificar que no tenga errores.
4. Si run_code devuelve error, corrígelo y vuelve a ejecutar.
5. Usa rag_query solo si la especificación lo requiere o necesitas contexto técnico.
6. Sé conciso. Entrega el código y un resumen corto de lo que creaste.
7. Crea agentes que sean modulares y fáciles de integrar después.

ESPECIFICACIÓN APROBADA:
{spec}

ARCHIVOS ACTUALES EN LA CARPETA:
{files}"""

Responde en este formato:
- Resumen corto
- Puntos clave
- Conclusión o recomendación

import os
from datetime import datetime
from langchain_ollama import ChatOllama

class GeneradorContenido:
    def __init__(self):
        self.llm = ChatOllama(
            model="qwen2.5:7b",
            temperature=0.7,
            num_ctx=8192,
        )
        self.carpeta = "contenido"
        os.makedirs(self.carpeta, exist_ok=True)

    def generar(self, tema: str, cantidad: int = 5) -> str:
        print(f"Generando ideas de contenido sobre: {tema}\n")

        prompt = f"""Eres un experto en contenido para TikTok e Instagram.
Genera {cantidad} ideas de posts, reels o stories sobre el siguiente tema.

Tema: {tema}

Para cada idea incluye:
- Tipo (Reel, Post, Story, Carrusel)
- Hook o primera frase
- Descripción breve
- Idea de engagement (pregunta, encuesta o llamado a la acción)
- Hashtags sugeridos

Sé creativo y enfocado en generar engagement y leads.
"""

        respuesta = self.llm.invoke(prompt)
        contenido = respuesta.content

        # Guardar
        nombre = "".join(c if c.isalnum() or c in " _-" else "_" for c in tema[:40])
        ruta = os.path.join(self.carpeta, f"{nombre}_{datetime.now().strftime('%Y%m%d_%H%M')}.txt")
        with open(ruta, "w", encoding="utf-8") as f:
            f.write(contenido)

        print(f"Ideas guardadas en: {ruta}\n")
        return contenido

if __name__ == "__main__":
    gen = GeneradorContenido()
    resultado = gen.generar("Crecimiento en TikTok e Instagram para generar leads")
    print("=" * 60)
    print(resultado)
    print("=" * 60)

import os
import json
from datetime import datetime
from langchain_ollama import ChatOllama

class GeneradorContenido:
    def __init__(self):
        self.llm = ChatOllama(
            model="qwen2.5:7b",
            temperature=0.7,
            num_ctx=8192,
        )
        self.carpeta = "contenido"
        self.historial_planes = "historial_planes.json"
        os.makedirs(self.carpeta, exist_ok=True)

    def cargar_ultimo_plan(self) -> str:
        """Carga el último plan aprobado."""
        if not os.path.exists(self.historial_planes):
            return ""
        try:
            with open(self.historial_planes, "r", encoding="utf-8") as f:
                planes = json.load(f)
            # Buscar el último aprobado
            for plan in reversed(planes):
                if plan.get("aprobado"):
                    return plan.get("plan", "")
            return ""
        except:
            return ""

    def generar(self, tema: str = "", cantidad: int = 5, usar_plan: bool = True) -> str:
        plan_contexto = ""
        if usar_plan:
            plan_contexto = self.cargar_ultimo_plan()
            if plan_contexto:
                print("Usando último plan aprobado como contexto.\n")

        tema_final = tema if tema else "Crecimiento en redes sociales y generación de leads"

        print(f"Generando ideas de contenido sobre: {tema_final}\n")

        prompt = f"""Eres un experto en contenido para TikTok e Instagram enfocado en crecimiento y leads.

Tema principal: {tema_final}

Plan de referencia (si existe):
{plan_contexto if plan_contexto else "Ninguno"}

Genera {cantidad} ideas de contenido. Para cada una incluye:
- Tipo (Reel, Carrusel, Story, Post, Live)
- Hook fuerte
- Descripción breve del contenido
- Engagement (pregunta, encuesta, CTA)
- Objetivo (alcance, leads, comunidad, etc.)
- Hashtags sugeridos

Haz que las ideas estén alineadas con el plan si se proporcionó uno.
"""

        respuesta = self.llm.invoke(prompt)
        contenido = respuesta.content

        # Guardar
        nombre = "".join(c if c.isalnum() or c in " _-" else "_" for c in tema_final[:40])
        ruta = os.path.join(self.carpeta, f"{nombre}_{datetime.now().strftime('%Y%m%d_%H%M')}.txt")
        with open(ruta, "w", encoding="utf-8") as f:
            f.write(contenido)

        print(f"Ideas guardadas en: {ruta}\n")

import os
import json
from datetime import datetime
from langchain_ollama import ChatOllama

class AgenteLeads:
    def __init__(self):
        self.llm = ChatOllama(
            model="qwen2.5:7b",
            temperature=0.6,
            num_ctx=8192,
        )
        self.carpeta = "leads"
        self.historial_path = "historial_leads.json"
        os.makedirs(self.carpeta, exist_ok=True)

    def generar_estrategia(self, nicho: str = "", objetivo: str = "") -> dict:
        print("Generando estrategia de leads...\n")

        prompt = f"""Eres un experto en generación de leads para redes sociales (TikTok e Instagram).

Nicho / Tema: {nicho if nicho else "Crecimiento personal o contenido en stream"}
Objetivo: {objetivo if objetivo else "Conseguir clientes potenciales"}

Genera una estrategia clara que incluya:
1. Formas de captar leads (lead magnets, comentarios, DMs, links, etc.)
2. Ejemplos de llamados a la acción (CTAs)
3. Ideas de contenido orientado a leads
4. Cómo organizar o dar seguimiento a los interesados
5. Posibles ofertas simples o siguientes pasos

Sé práctico y directo.
"""

        respuesta = self.llm.invoke(prompt)
        estrategia = respuesta.content

        resultado = {
            "fecha": datetime.now().strftime("%Y-%m-%d %H:%M"),
            "nicho": nicho,
            "objetivo": objetivo,
            "estrategia": estrategia
        }

        # Guardar archivo
        nombre = "".join(c if c.isalnum() or c in " _-" else "_" for c in (nicho or "leads")[:40])
        ruta = os.path.join(self.carpeta, f"{nombre}_{datetime.now().strftime('%Y%m%d_%H%M')}.txt")
        with open(ruta, "w", encoding="utf-8") as f:
            f.write(estrategia)

        # Historial
        historial = []
        if os.path.exists(self.historial_path):
            try:
                with open(self.historial_path, "r", encoding="utf-8") as f:
                    historial = json.load(f)
            except:
                historial = []
        historial.append(resultado)
        with open(self.historial_path, "w", encoding="utf-8") as f:
            json.dump(historial, f, ensure_ascii=False, indent=2)

        print(f"Estrategia guardada en: {ruta}\n")
        print("=" * 60)
        print(estrategia)
        print("=" * 60)

        return resultado

if __name__ == "__main__":
    agente = AgenteLeads()
    agente.generar_estrategia(
        nicho="Contenido en stream y redes sociales",
        objetivo="Conseguir leads y clientes potenciales para una futura feria o producto"
    )

import json
from datetime import datetime
from langchain_ollama import ChatOllama

class AgenteLeads:
    def __init__(self):
        self.llm = ChatOllama(
            model="qwen2.5:7b",
            temperature=0.6,
            num_ctx=8192,
        )
        self.carpeta = "leads"
        self.historial_path = "historial_leads.json"
        self.historial_planes = "historial_planes.json"
        os.makedirs(self.carpeta, exist_ok=True)

    def cargar_ultimo_plan(self) -> str:
        if not os.path.exists(self.historial_planes):
            return ""
        try:
            with open(self.historial_planes, "r", encoding="utf-8") as f:
                planes = json.load(f)
            for plan in reversed(planes):
                if plan.get("aprobado"):
                    return plan.get("plan", "")[:1500]
            return ""
        except:
            return ""

    def generar_estrategia(self, nicho: str = "", objetivo: str = "") -> dict:
        print("Generando estrategia de leads...\n")
        plan = self.cargar_ultimo_plan()

        prompt = f"""Eres un experto en generación de leads para TikTok e Instagram.

Nicho: {nicho or "Contenido y stream"}
Objetivo: {objetivo or "Conseguir clientes potenciales"}

Plan de referencia:
{plan if plan else "Ninguno"}

Genera una estrategia práctica con:
1. Formas de captar leads
2. CTAs concretos
3. Ideas de contenido para leads
4. Cómo dar seguimiento
5. Oferta o siguiente paso simple
"""

        respuesta = self.llm.invoke(prompt)
        estrategia = respuesta.content

        resultado = {
            "fecha": datetime.now().strftime("%Y-%m-%d %H:%M"),
            "nicho": nicho,
            "objetivo": objetivo,
            "estrategia": estrategia
        }

        nombre = "".join(c if c.isalnum() or c in " _-" else "_" for c in (nicho or "leads")[:40])
        ruta = os.path.join(self.carpeta, f"{nombre}_{datetime.now().strftime('%Y%m%d_%H%M')}.txt")
        with open(ruta, "w", encoding="utf-8") as f:
            f.write(estrategia)

        historial = []
        if os.path.exists(self.historial_path):
            try:
                with open(self.historial_path, "r", encoding="utf-8") as f:
                    historial = json.load(f)
            except:
                pass
        historial.append(resultado)
        with open(self.historial_path, "w", encoding="utf-8") as f:
            json.dump(historial, f, ensure_ascii=False, indent=2)

        print(f"Estrategia guardada en: {ruta}\n")
        print("=" * 60)
        print(estrategia)
        print("=" * 60)
        return resultado

if __name__ == "__main__":
    AgenteLeads().generar_estrategia(
        nicho="Contenido en stream y redes sociales",
        objetivo="Leads y clientes potenciales"
    )

import os
import json
from datetime import datetime
from langchain_ollama import ChatOllama

class AgenteEngagement:
    def __init__(self):
        self.llm = ChatOllama(
            model="qwen2.5:7b",
            temperature=0.7,
            num_ctx=8192,
        )
        self.carpeta = "engagement"
        self.historial_path = "historial_engagement.json"
        os.makedirs(self.carpeta, exist_ok=True)

    def sugerir_respuestas(self, comentario: str, contexto: str = "") -> dict:
        print("Generando sugerencias de respuesta...\n")

        prompt = f"""Eres un experto en engagement para TikTok e Instagram.
Tu estilo es cercano, natural y orientado a generar conversación o leads.

Comentario del usuario:
"{comentario}"

Contexto adicional (opcional):
{contexto if contexto else "Ninguno"}

Genera 3 posibles respuestas:
1. Una respuesta amigable que invite a seguir conversando
2. Una respuesta que haga una pregunta
3. Una respuesta con un llamado a la acción suave (lead o siguiente paso)

Sé breve, natural y evita sonar robótico.
"""

        respuesta = self.llm.invoke(prompt)
        sugerencias = respuesta.content

        resultado = {
            "fecha": datetime.now().strftime("%Y-%m-%d %H:%M"),
            "comentario": comentario,
            "contexto": contexto,
            "sugerencias": sugerencias
        }

        # Guardar
        nombre = "".join(c if c.isalnum() or c in " _-" else "_" for c in comentario[:30])
        ruta = os.path.join(self.carpeta, f"{nombre}_{datetime.now().strftime('%Y%m%d_%H%M')}.txt")
        with open(ruta, "w", encoding="utf-8") as f:
            f.write(f"Comentario: {comentario}\n\n{sugerencias}")

        # Historial
        historial = []
        if os.path.exists(self.historial_path):
            try:
                with open(self.historial_path, "r", encoding="utf-8") as f:
                    historial = json.load(f)
            except:
                pass
        historial.append(resultado)
        with open(self.historial_path, "w", encoding="utf-8") as f:
            json.dump(historial, f, ensure_ascii=False, indent=2)

        print(f"Guardado en: {ruta}\n")
        print("=" * 60)
        print(sugerencias)
        print("=" * 60)

        return resultado

    def generar_ideas_engagement(self, tema: str) -> str:
        """Genera ideas generales de engagement para un tema."""
        prompt = f"""Genera 5 ideas de engagement para contenido sobre: {tema}

Incluye:
- Preguntas para comentarios
- Ideas de encuestas
- CTAs suaves
- Formas de incentivar respuestas
"""
        respuesta = self.llm.invoke(prompt)
        return respuesta.content

if __name__ == "__main__":
    agente = AgenteEngagement()
    
    # Ejemplo de uso
    agente.sugerir_respuestas(
        comentario="¿Cómo empezaste con esto?",
        contexto="Contenido sobre crecimiento en redes y stream"
    )

from agente_investigación import Investigador
from agente_planificador import Planificador
from agente_contenido import GeneradorContenido
from agente_leads import AgenteLeads
from agente_engagement import AgenteEngagement

class Coordinador:
    def __init__(self):
        self.investigador = Investigador()
        self.planificador = Planificador()
        self.generador = GeneradorContenido()
        self.leads = AgenteLeads()
        self.engagement = AgenteEngagement()

    def ejecutar(self, tema: str, objetivo: str):
        print(f"=== Iniciando proceso para: {tema} ===\n")

        # 1. Investigar
        print("1. Investigando...")
        investigacion = self.investigador.investigar(tema)
        info = investigacion.get("respuesta", "")

        # 2. Planificar
        print("\n2. Generando plan...")
        plan = self.planificador.crear_plan(
            informacion=info,
            objetivo=objetivo,
            usar_investigaciones=True
        )

        if not plan.get("aprobado"):
            print("\nPlan no aprobado. Proceso detenido.")
            return plan

        # 3. Contenido
        print("\n3. Generando ideas de contenido...\n")
        self.generador.generar(tema)

        # 4. Leads
        print("\n4. Generando estrategia de leads...\n")
        self.leads.generar_estrategia(nicho=tema, objetivo=objetivo)

        # 5. Ideas de engagement
        print("\n5. Generando ideas de engagement...\n")
        ideas_eng = self.engagement.generar_ideas_engagement(tema)
        print(ideas_eng)

        print("\n=== Proceso terminado ===")
        return plan

if __name__ == "__main__":
    coord = Coordinador()
    coord.ejecutar(
        tema="Tácticas de engagement y crecimiento en TikTok e Instagram para generar leads",
        objetivo="Crecer en redes sociales para obtener leads y clientes potenciales"
    )

import os
import json
from datetime import datetime

def generar_resumen():
    archivos = [
        "historial_investigaciones.json",
        "historial_planes.json",
        "historial_leads.json",
        "historial_engagement.json"
    ]
    
    resumen = f"Resumen del sistema - {datetime.now().strftime('%Y-%m-%d %H:%M')}\n"
    resumen += "=" * 50 + "\n\n"

    for archivo in archivos:
        resumen += f"## {archivo}\n"
        if os.path.exists(archivo):
            try:
                with open(archivo, "r", encoding="utf-8") as f:
                    data = json.load(f)
                resumen += f"Entradas: {len(data)}\n"
                if data:
                    ultimo = data[-1]
                    resumen += f"Último: {str(ultimo)[:300]}...\n"
            except Exception as e:
                resumen += f"Error leyendo: {e}\n"
        else:
            resumen += "No existe aún.\n"
        resumen += "\n"

    with open("resumen_sistema.txt", "w", encoding="utf-8") as f:
        f.write(resumen)
    
    print(resumen)
    print("Resumen guardado en: resumen_sistema.txt")

if __name__ == "__main__":
    generar_resumen()

class Companero:

        self.historial = []

    def hablar(self, mensaje: str) -> str:

        texto = respuesta.content
        self.historial.append({"role": "assistant", "content": texto})
        return texto

Entendido. La idea es que cada agente tenga un framework / guía clara basada en investigación, para que pueda trabajar de forma más autónoma sin depender de memoria conversacional.

Cómo lo podemos hacer
El investigador genera un framework específico para un rol (por ejemplo “framework para editar clips” o “framework para responder emails”).
Ese framework se guarda en un archivo.
Los agentes especializados lo leen al iniciar y lo usan como base para trabajar.
Ejemplo práctico: agente_framework.py
Este agente crea frameworks para los demás:

python
import os
from datetime import datetime
from langchain_ollama import ChatOllama

class AgenteFramework:
    def __init__(self):
        self.llm = ChatOllama(
            model="qwen2.5:7b",
            temperature=0.4,
            num_ctx=4096,
            num_predict=1200,
        )
        self.carpeta = "frameworks"
        os.makedirs(self.carpeta, exist_ok=True)

    def crear_framework(self, rol: str, objetivo: str) -> str:
        prompt = f"""Crea un framework claro y práctico para un agente que se dedica a: {rol}

Objetivo del agente: {objetivo}

El framework debe incluir:
- Principios clave
- Pasos o proceso recomendado
- Qué debe priorizar
- Qué debe evitar
- Criterios de buena calidad
- Formato de salida sugerido

Hazlo concreto y usable por otro agente.
"""
        respuesta = self.llm.invoke(prompt)
        contenido = respuesta.content

        nombre = "".join(c if c.isalnum() or c in " _-" else "_" for c in rol[:40])
        ruta = os.path.join(self.carpeta, f"framework_{nombre}.txt")
        with open(ruta, "w", encoding="utf-8") as f:
            f.write(contenido)

        print(f"Framework guardado en: {ruta}")
        return contenido

if __name__ == "__main__":
    af = AgenteFramework()
    print(af.crear_framework(
        rol="Editor de clips de stream",
        objetivo="Crear clips cortos, atractivos y con buen hook para TikTok y Shorts"
    ))

        return "Sin framework. Actúa de forma práctica y clara."

    def procesar(self, descripcion: str) -> str:

Explicar el sesgo de confirmación
Sesgo de confirmación (confirmation bias) es la tendencia a buscar, interpretar, recordar y dar más peso a la información que confirma lo que ya creemos, mientras se ignora o se minimiza la información que lo contradice.

Cómo funciona
Búsqueda selectiva: Prefieres fuentes, videos, artículos o personas que digan cosas parecidas a lo que ya piensas.
Interpretación selectiva: Cuando ves datos ambiguos, los lees de la forma que más encaja con tu idea previa.
Memoria selectiva: Recuerdas mejor los ejemplos que te dan la razón y olvidas o restas importancia a los que no.
Evitar la disonancia: La información que choca con tus creencias genera incomodidad, así que el cerebro tiende a descartarla o a explicarla de forma que no te obligue a cambiar de opinión.
Ejemplos simples
Alguien convencido de que una dieta funciona solo recuerda los días en que se sintió bien y olvida los que no.

En una discusión, dos personas ven la misma noticia y cada una concluye que “confirma” su postura.

No es que la gente sea “tonta”; es un atajo mental muy común. Reduce el esfuerzo de reevaluar creencias, pero puede llevar a decisiones peores, polarización y a sobreestimar cuánta evidencia real apoya lo que pensamos.
