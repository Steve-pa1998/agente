import imaplib, email, smtplib, json, os, requests
from email.mime.text import MIMEText

from datetime import datetime

def load_env():
    env = {}
    for line in (Path(__file__).parent.parent / ".env").read_text().strip().splitlines():

    return env

def connect():
    mail = imaplib.IMAP4_SSL("imap.gmail.com")

def get_unread(mail):
    status, data = mail.search(None, "UNSEEN")
    ids = data[0].split()
    out = []
    for id_ in ids:
        status, msg_data = mail.fetch(id_, "(RFC822)")
        msg = email.message_from_bytes(msg_data[0][1])
        body = ""
        if msg.is_multipart():
            for part in msg.walk():
                if part.get_content_type() == "text/plain":
                    body = part.get_payload(decode=True).decode("utf-8", errors="ignore")
                    break
        else:
            p = msg.get_payload(decode=True)
            if p: body = p.decode("utf-8", errors="ignore")
        out.append({"id": id_.decode(), "from": str(msg["From"]), "subject": str(msg["Subject"]), "date": str(msg["Date"]), "body": body})
    return out

def classify(e):
    if not GROQ_KEY:
        return {"action": "flag", "priority": "media", "reply": None}

Responde SOLO en JSON:
{{"action": "reply"|"flag"|"ignore", "priority": "alta"|"media"|"baja", "reply": "texto o null"}}
De: {e['from']}
Asunto: {e['subject']}
Cuerpo: {e['body'][:500]}"""
    try:
        r = requests.post(GROQ_URL, headers={"Authorization": f"Bearer {GROQ_KEY}"},

        return json.loads(r.json()["choices"][0]["message"]["content"])

Modificación genética de la marihuana — cómo funciona (lo real, no la hype): 

Lo que ya existe y se usa hoy:

Técnica	Qué hace	Estado
Cruce selectivo (F1)	Cruzas 2 plantas estables → híbrido uniforme	Lo que hace el 99% de los breeders. Lento pero funciona.
Single Seed Descent (SSD)	Auto-polinizas una planta generación tras generación hasta que es 100% homozygosa	Da líneas "puras" para usar como padres.  Muy lento (5-6 generaciones).
Triploides	Duplicas cromosomas con colchicina (4x) → cruzas con normal (2x) → nace 3x	Reduce drásticamente la producción de semillas. Se usa para "semillas estériles" comerciales.

Selección genómica (GS)	Mides miles de SNPs (variaciones en el ADN) y estimas el "valor de cría" de la planta para rasgos complejos (aroma, rendimiento, vigor)	Como un "score genético" antes de que la planta crezca. 

Lo nuevo (2024-2026):

Técnica	Qué hace	Estado
CRISPR/Cas9	Cortas un gen específico (ej: THCAS, CBDAS, o un gen de resistencia) y lo apagas o modificas	Ya se hizo en hemp (Wisconsin Crop Innovation Center). USDA dijo que NO es "biotech regulado" (2024-2026).  Es decir: legalmente no lo consideran OGM.

Epigenómica (jul 2026)	No cambias el ADN, cambias cómo se lee. dCas9 + metiltransferasas para "encender" o "apagar" genes de cannabinoides sin editar la secuencia	Aún en fase de diseño experimental. Pero si funciona, es más limpio que CRISPR (no dejas huella en el ADN).

Lo que controla el THC/CBD (el gen real):

No es un solo gen. Son 2 enzimas que compiten por el mismo sustrato:

THCAS → produce THC
CBDAS → produce CBD 
Están en una región del genoma llena de retrotransposones y copias pseudogenes (más de 30 copias muertas). Si THCAS está activo y CBDAS está apagado → planta de THC. Si al revés → CBD. Si ambos activos → mixto. 

Lo que CRISPR puede hacer con esto:

Apagar CBDAS → 100% THC (ya se hace por cruce, pero CRISPR lo hace en 1 generación en vez de 5)
Apagar THCAS → 100% CBD (hemp industrial)
Editar genes de terpenos → cambiar el aroma (limón, tierra, menta) sin cambiar el THC
Editar genes de resistencia → hongo, sequía, salinidad
Lo que NO se puede hacer (aún):

No puedes "diseñar" una planta desde cero. CRISPR edita lo que ya está. No añade genes de otras especies (eso sería transgénico, y en cannabis es mucho más difícil por el problema de regeneración).
El genoma es enorme (~900 Mb, 74% repetitivo). No es como editar una bacteria. Es como buscar una aguja en un heno del tamaño de un edificio.
La regeneración (hacer que la planta editada vuelva a crecer completa) sigue siendo el cuello de botella. En 2025 se logró un protocolo que funciona en 70-90% de las cepas, pero no en todas.
En resumen: La marihuana ya se "modifica genéticamente" desde que existe la agricultura (cruce selectivo = modificación genética, solo que lenta). Lo nuevo es que ahora puedes hacerlo preciso y rápido (CRISPR) en vez de lento y a ciegas (cruce). Y en 3 años va a estar más avanzado de lo que está hoy.

import subprocess, os, json, sys
from pathlib import Path
from datetime import datetime

MUSICA_DIR = Path(__file__).parent.parent / "musica"
RAW_DIR = MUSICA_DIR / "raw"
OUT_DIR = MUSICA_DIR / "tracks"
RAW_DIR.mkdir(parents=True, exist_ok=True)
OUT_DIR.mkdir(parents=True, exist_ok=True)

def load_env():
    env = {}
    for line in (Path(__file__).parent.parent / ".env").read_text().strip().splitlines():
        if "=" in line and not line.startswith("#"):
            k, v = line.split("=", 1)
            env[k.strip()] = v.strip()
    return env

env = load_env()
GROQ_KEY = env.get("GROQ_API_KEY", "")

def list_raw():
    """Lista grabaciones sin procesar."""
    files = list(RAW_DIR.glob("*.wav")) + list(RAW_DIR.glob("*.mp3"))
    files.sort(key=lambda x: x.stat().st_mtime, reverse=True)
    return files

def process(input_file, output_name=None):
    """
    Procesa una grabación:
    1. Normaliza volumen
    2. Aplica compresión suave
    3. Exporta MP3
    Nota: El autotune (Graillon/MAutoPitch) se aplica en Cakewalk Sonar,
    no por script. Este script hace pre-producción.
    """
    input_path = Path(input_file)
    if not input_path.exists():
        print(f"[ERROR] No existe: {input_file}")
        return

    if not output_name:
        output_name = input_path.stem
    out_path = OUT_DIR / f"{output_name}_{datetime.now():%Y%m%d}.mp3"

    # Normalizar + compresión + exportar MP3
    cmd = [
        "ffmpeg", "-y",
        "-i", str(input_path),
        "-af", "loudnorm=I=-14:TP=-1.5:LRA=11,acompressor=threshold=-20dB:ratio=3:attack=5:release=100",
        "-b:a", "192k",
        str(out_path)
    ]

    print(f"[PROD] Procesando: {input_path.name}")
    r = subprocess.run(cmd, capture_output=True, text=True)
    if r.returncode == 0:
        size = out_path.stat().st_size // 1024
        print(f"[PROD] ✅ {out_path.name} ({size} KB)")
        return str(out_path)
    else:
        print(f"[PROD] ❌ Error: {r.stderr[:200]}")
        return None

def generate_portada(titulo, estilo="dark"):
    """Genera una portada simple con ffmpeg (colores + texto)."""
    out_path = OUT_DIR / f"portada_{titulo.replace(' ','_')}.png"
    # Fondo oscuro con texto
    cmd = [
        "ffmpeg", "-y",
        "-f", "lavfi", "-i", f"color=c=0x1a1a2e:s=1000x1000:d=1",
        "-vf", f"drawtext=text='{titulo}':fontcolor=white:fontsize=72:x=(w-text_w)/2:y=(h-text_h)/2",
        "-frames:v", "1",
        str(out_path)
    ]
    r = subprocess.run(cmd, capture_output=True, text=True)
    if r.returncode == 0:
        print(f"[PORTADA] ✅ {out_path.name}")
    return str(out_path)

def main():
    if len(sys.argv) > 1:
        cmd = sys.argv[1]
        if cmd == "list":
            files = list_raw()
            if not files:
                print("No hay grabaciones en musica/raw/")
            for f in files:
                print(f"  • {f.name} ({f.stat().st_size//1024} KB) — {datetime.fromtimestamp(f.stat().st_mtime):%d/%m %H:%M}")
        elif cmd == "process":
            if len(sys.argv) > 2:
                process(sys.argv[2])
            else:
                files = list_raw()
                if files:
                    process(files[0])
                else:
                    print("No hay grabaciones para procesar.")
        elif cmd == "portada":
            titulo = " ".join(sys.argv[2:]) if len(sys.argv) > 2 else "Track"
            generate_portada(titulo)
        else:
            print("Uso: python agente_produccion.py [list|process|portada]")
    else:
        print("=== AGENTE PRODUCCIÓN ===")
        print("Comandos:")
        print("  list     — ver grabaciones")
        print("  process  — procesar la más reciente (normalizar + comprimir)")
        print("  portada  — generar portada")
        print("")
        print("Para autotune: abre Cakewalk Sonar → arrastra Graillon/MAutoPitch a la pista")

if __name__ == "__main__":
    main()

Investigación de software de audio y alternativas
Guía rápida de cada programa
Cakewalk Sonar (DAW — donde grabas y mezclas)
Lo que tienes gratis:

Pistas de audio/MIDI: ilimitadas
ProChannel (equalizador, compresor, reverb): sí, pero con módulos limitados
AudioSnap (cortar/pegar audio visualmente): sí
VST2 + VST3: soporte completo
Arranger tracks: solo 1 (para organizar secciones)
Lo que NO tienes (solo con suscripción):

Cakewalk Assist (IA)
Core Plugin Suite
TH-U amp sim completo
Over-sampling de plugins
Track Manager
Cómo se usa (flujo básico):

Paso	Qué haces
1. Instalar	cakewalk.com/sonar → cuenta BandLab → Cakewalk Product Center → Install
2. Primer arranque	Se abre → te pide suscripción → "Continue with free license" → espera el plugin scan
3. Verificar VSTs	Edit → Preferences → File → VST Settings → verifica que C:\Program Files\Common Files\VST3 está en la lista → si no, Add → OK
4. Grabar	Track → Add Track → Audio → selecciona tu mic en Input → click en el botón rojo (Arm) → click en Record (o R) → cantas → click en Stop
5. Agregar autotune	Click en la pista de audio → Insert FX (o click en el slot de FX) → busca Graillon 3 o MAutoPitch → OK
6. Ajustar	Abre el plugin (click en el nombre) → ajusta (abajo) → play
7. Exportar	File → Export → Audio → MP3 o WAV → guardas en musica\raw\

Atajos útiles:

R = Record
Espacio = Play/Stop
Ctrl+Z = Deshacer
Ctrl+S = Guardar
Ctrl+A = Seleccionar todo 
Graillon 3 Free (Autotune — el mejor gratis)
Controles principales:

Control	Qué hace	Valor recomendado
Correction (ON/OFF)	Activa/desactiva la corrección	ON
Correction Amount	Cuánto corrige (0 = nada, 100 = todo)	100 (efecto autotune) o 50 (natural)
Smooth	Velocidad de transición (mínimo = robot, máximo = suave)	Mínimo para T-Pain, 50-100 para natural
Inertia	Hace que las notas "se peguen" (evita saltos raros)	20-40
Scale (tecla de solfeo)	Elige Major o Minor	Según tu canción
Keyboard (piano abajo)	Qué notas están permitidas	Apaga las que NO estén en tu tonalidad
Pitch Shift	Sube/baja tu voz (semitonos)	0 (normal) o ±12 para cambiar registro
Formant	Mantiene tu voz "humana" al hacer pitch shift	ON si usas pitch shift

Cómo se usa (paso a paso):

Graba tu voz en Cakewalk
Inserta Graillon 3 en la pista (Insert FX → Graillon 3)
Elige la escala: Click en el icono de clave (arriba del piano) → Major o Minor
Configura el piano: Si tu canción está en Do Mayor (C Major) → apaga las 5 teclas negras.  Si está en La menor (A Minor) → también apagas las 5 negras (misma escala).  Si no sabes la tonalidad → deja todas encendidas (modo cromático)
Ajusta:
Efecto "robot" (T-Pain): Correction Amount = 100, Smooth = mínimo
Corrección natural: Correction Amount = 50-70, Smooth = 50-100
Invisible (solo te salva de las notas feas): Correction Amount = 30, Smooth = 100
Play → escucha → ajusta 
Tip: Si una nota se corrige a la nota "equivocada" (cambia la melodía), es que la escala no está bien. Prueba la otra (Major ↔ Minor).

MAutoPitch (Melda — alternativa)
Controles principales:

Control	Qué hace	Valor recomendado
Depth	Cuánto corrige (0 = nada, 100 = todo)	100 (autotune) o 30-50 (transparente)
Speed	Velocidad de corrección (bajo = robot, alto = suave)	10-30 (robot) o 100+ (natural)
Detune	Desafina a propósito (efecto)	0
Formant Shift	Cambia el "color" de tu voz sin cambiar notas	0 (normal)
Keep Formants	Mantiene tu voz natural al hacer pitch shift	ON
Width	Expansión estéreo (hace que suene a "doble voz")	0-30
Threshold	Cuánto tiene que estar "feita" la nota para corregirla	-20 a -30 dB
Stabilization	Evita "wobble" (temblor)	10-30
